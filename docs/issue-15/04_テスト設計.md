# テスト設計

## テスト方針

- **単体テスト**: Vitest で hooks・ store・ユーティリティ関数をテスト
- **コンポーネントテスト**: React Testing Library で ページコンポーネント・レイアウトをテスト
- **テストカバレッジ**: 主要なユーザーフロー + エラーケース + バリデーション
- **モック**: MSW で API をモック、sessionStorage は localStorage テストダブルで代替

## モック設定（MSW）

### handlers/auth.mock.ts

```typescript
import { http, HttpResponse } from 'msw';
import { API_PATHS } from '@/lib/constants/api-paths';

const BASE_URL = import.meta.env.VITE_API_URL;

export const authHandlers = [
  http.post(`${BASE_URL}/api${API_PATHS.AUTH.LOGIN}`, async ({ request }) => {
    const body = await request.json();
    if (body.username === 'success' && body.password === 'pass') {
      return HttpResponse.json({
        loginStatus: 'SUCCESS',
        id: '1',
        name: 'Test User',
        role: 'USER',
        token: 'mock-token',
        groups: [],
      });
    }
    if (body.username === 'reset' && body.password === 'pass') {
      return HttpResponse.json({
        loginStatus: 'REQUIRES_PASSWORD_RESET',
        id: '2',
        name: 'Reset User',
        role: 'USER',
        reason: 'INITIAL',
      });
    }
    return HttpResponse.json({ message: 'Invalid credentials' }, { status: 401 });
  }),

  http.get(`${BASE_URL}/api${API_PATHS.AUTH.SESSION}`, () => {
    return HttpResponse.json({
      id: '1',
      name: 'Test User',
      role: 'USER',
      tenant_id: 'tenant-1',
    });
  }),

  http.post(`${BASE_URL}/api${API_PATHS.AUTH.LOGOUT}`, () => {
    return HttpResponse.text('OK');
  }),

  http.post(`${BASE_URL}/api${API_PATHS.AUTH.PASSWORD_RESET}`, async ({ request }) => {
    const body = await request.json();
    if (body.newPassword.length < 8) {
      return HttpResponse.json(
        { message: 'Password too short' },
        { status: 400 }
      );
    }
    return HttpResponse.json({
      id: '2',
      name: 'Reset User',
      role: 'USER',
      token: 'new-token',
      groups: [],
    });
  }),
];
```

## テスト環境構成

```typescript
// vitest.config.ts
import { defineConfig } from 'vitest/config';
import react from '@vitejs/plugin-react';
import path from 'path';

export default defineConfig({
  plugins: [react()],
  test: {
    environment: 'jsdom',
    globals: true,
    setupFiles: ['./src/test/setup.ts'],
  },
  resolve: {
    alias: {
      '@': path.resolve(__dirname, './src'),
    },
  },
});
```

```typescript
// src/test/setup.ts
import { expect, afterEach, vi } from 'vitest';
import { cleanup } from '@testing-library/react';
import { setupServer } from 'msw/node';
import { authHandlers } from '@/app/core/mocks/handlers/auth.mock';

const server = setupServer(...authHandlers);

beforeAll(() => server.listen());
afterEach(() => {
  cleanup();
  server.resetHandlers();
});
afterAll(() => server.close());

// localStorage mock
const localStorageMock = (() => {
  let store: Record<string, string> = {};
  return {
    getItem: (key: string) => store[key] || null,
    setItem: (key: string, value: string) => {
      store[key] = value.toString();
    },
    removeItem: (key: string) => {
      delete store[key];
    },
    clear: () => {
      store = {};
    },
  };
})();

Object.defineProperty(window, 'sessionStorage', {
  value: localStorageMock,
});
```
