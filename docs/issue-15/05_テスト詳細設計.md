## `src/hooks/use-auth.test.ts`

```typescript
import { describe, test, expect, beforeEach, vi } from 'vitest';
import { renderHook, waitFor } from '@testing-library/react';
import { QueryClient, QueryClientProvider } from '@tanstack/react-query';
import { useLoginMutation, useResetPasswordMutation, useLogoutMutation, useSessionQuery } from './use-auth';

const queryClient = new QueryClient({
  defaultOptions: {
    queries: { retry: false },
    mutations: { retry: false },
  },
});

const wrapper = ({ children }: { children: React.ReactNode }) => (
  <QueryClientProvider client={queryClient}>{children}</QueryClientProvider>
);

describe('useLoginMutation', () => {
  test('ログイン成功時に SUCCESS を返すこと', async () => {
    const { result } = renderHook(() => useLoginMutation(), { wrapper });

    result.current.mutate({ username: 'success', password: 'pass' });

    await waitFor(() => {
      expect(result.current.isSuccess).toBe(true);
      expect(result.current.data?.loginStatus).toBe('SUCCESS');
    });
  });

  test('パスワードリセット必須時に REQUIRES_PASSWORD_RESET を返すこと', async () => {
    const { result } = renderHook(() => useLoginMutation(), { wrapper });

    result.current.mutate({ username: 'reset', password: 'pass' });

    await waitFor(() => {
      expect(result.current.isSuccess).toBe(true);
      expect(result.current.data?.loginStatus).toBe('REQUIRES_PASSWORD_RESET');
    });
  });

  test('ログイン失敗時にエラーをスローすること', async () => {
    const { result } = renderHook(() => useLoginMutation(), { wrapper });

    result.current.mutate({ username: 'invalid', password: 'wrong' });

    await waitFor(() => {
      expect(result.current.isError).toBe(true);
    });
  });
});

describe('useResetPasswordMutation', () => {
  test('パスワードリセット成功時に AuthSessionResponse を返すこと', async () => {
    const { result } = renderHook(() => useResetPasswordMutation(), { wrapper });

    result.current.mutate({
      username: 'user',
      oldPassword: 'old',
      newPassword: 'NewPass123',
    });

    await waitFor(() => {
      expect(result.current.isSuccess).toBe(true);
      expect(result.current.data?.id).toBeDefined();
      expect(result.current.data?.name).toBeDefined();
    });
  });

  test('パスワード短い時にエラーをスローすること', async () => {
    const { result } = renderHook(() => useResetPasswordMutation(), { wrapper });

    result.current.mutate({
      username: 'user',
      oldPassword: 'old',
      newPassword: 'short',
    });

    await waitFor(() => {
      expect(result.current.isError).toBe(true);
    });
  });
});

describe('useLogoutMutation', () => {
  test('ログアウト成功時に 成功を返すこと', async () => {
    const { result } = renderHook(() => useLogoutMutation(), { wrapper });

    result.current.mutate();

    await waitFor(() => {
      expect(result.current.isSuccess).toBe(true);
    });
  });
});

describe('useSessionQuery', () => {
  test('セッション有効時に BackendAuthResponse を返すこと', async () => {
    const { result } = renderHook(() => useSessionQuery(), { wrapper });

    await waitFor(() => {
      expect(result.current.isSuccess).toBe(true);
      expect(result.current.data?.id).toBeDefined();
      expect(result.current.data?.tenant_id).toBeDefined();
    });
  });
});
```

import { describe, test, expect, beforeEach, afterEach, vi } from 'vitest';
import { act } from '@testing-library/react';
import { useAuthStore } from './auth-store';

describe('useAuthStore', () => {
  beforeEach(() => {
    // sessionStorage をリセット
    sessionStorage.clear();
    // store をリセット（必要な場合）
  });

  test('初期状態で user は null であること', () => {
    const store = useAuthStore.getState();
    expect(store.user).toBe(null);
  });

  test('ensureInitialized で getSession が呼ばれること', async () => {
    const store = useAuthStore.getState();
    await store.ensureInitialized();

    expect(store.user).not.toBe(null);
    expect(store.user?.id).toBeDefined();
  });

  test('ensureInitialized は2回目以降、キャッシュを使用すること', async () => {
    const store = useAuthStore.getState();

    await store.ensureInitialized();
    const firstUser = store.user;

    await store.ensureInitialized();
    const secondUser = store.user;

    expect(firstUser).toBe(secondUser);
  });

  test('login 成功時に user がセットされること', async () => {
    const store = useAuthStore.getState();

    await act(async () => {
      await store.login({ username: 'success', password: 'pass' });
    });

    expect(store.user).not.toBe(null);
    expect(store.user?.id).toBe('1');
    expect(store.user?.name).toBe('Test User');
  });

  test('login 成功時に sessionStorage に user が保存されること', async () => {
    const store = useAuthStore.getState();

    await act(async () => {
      await store.login({ username: 'success', password: 'pass' });
    });

    const stored = sessionStorage.getItem('AUTH_USER');
    expect(stored).not.toBeNull();
    const parsed = JSON.parse(stored!);
    expect(parsed.id).toBe('1');
  });

  test('login パスワードリセット必須時に user が null になること', async () => {
    const store = useAuthStore.getState();

    expect(store.user).toBe(null);
  });

  test('logout 成功時に user が null になること', async () => {

    // まず login
    await act(async () => {
      await store.login({ username: 'success', password: 'pass' });
    });
    expect(store.user).not.toBe(null);

    // logout
    await act(async () => {
      await store.logout();
    });

    expect(store.user).toBe(null);
  });
