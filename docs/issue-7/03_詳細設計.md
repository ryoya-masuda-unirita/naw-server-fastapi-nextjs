### 4. `backend/app/core/security.py`（新規）

```python
from datetime import datetime, timedelta
from typing import Optional
import jwt
from passlib.context import CryptContext
from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPBearer, HTTPAuthCredentials
from sqlalchemy.orm import Session
from backend.app.database import get_db
from backend.app.models.user import User
from backend.app.models.password_history import PasswordHistory

# JWT 設定
SECRET_KEY = "your-secret-key"  # 本来は環境変数から読込
ALGORITHM = "HS512"
ACCESS_TOKEN_EXPIRE_HOURS = 5

# パスワードハッシュ化設定
pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")
security = HTTPBearer()

def hash_password(password: str) -> str:
    """パスワードを bcrypt ハッシュ化"""
    return pwd_context.hash(password)

def verify_password(plain_password: str, hashed_password: str) -> bool:
    """パスワードを検証"""
    return pwd_context.verify(plain_password, hashed_password)

def create_access_token(login_id: str, tenant_id: str) -> str:
    """JWT トークンを生成"""
    expire = datetime.utcnow() + timedelta(hours=ACCESS_TOKEN_EXPIRE_HOURS)
    payload = {
        "sub": login_id,
        "tenantId": tenant_id,
        "exp": expire,
        "iat": datetime.utcnow(),
    }
    encoded_jwt = jwt.encode(payload, SECRET_KEY, algorithm=ALGORITHM)
    return encoded_jwt

def decode_token(token: str) -> dict:
    """JWT トークンをデコード"""
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        return payload
    except jwt.ExpiredSignatureError:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Token expired")
    except jwt.InvalidTokenError:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Invalid token")

async def get_current_user(
    credentials: HTTPAuthCredentials = Depends(security),
    db: Session = Depends(get_db)
) -> User:
    """認証済みユーザーを取得（/api/** の保護に使用）"""
    token = credentials.credentials
    payload = decode_token(token)
    login_id = payload.get("sub")
    
    if not login_id:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Invalid token")
    
    user = db.query(User).filter(User.login_id == login_id).first()
    if not user:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="User not found")
    
    return user

def get_tenant_id_from_header(request) -> str:
    """リクエストヘッダーから X-Tenant-ID を取得"""
    tenant_id = request.headers.get("X-Tenant-ID")
    if not tenant_id:
        raise HTTPException(status_code=status.HTTP_400_BAD_REQUEST, detail="X-Tenant-ID header is required")
    return tenant_id
```

### 6. `backend/app/services/password_service.py`（新規）

```python
from datetime import datetime
from sqlalchemy.orm import Session
from fastapi import HTTPException, status
from backend.app.models.user import User
from backend.app.models.password_history import PasswordHistory
from backend.app.models.tenant import Tenant
from backend.app.core.security import hash_password, verify_password

class PasswordService:
    @staticmethod
    def get_current_password_hash(user_id: str, db: Session) -> str:
        """最新のパスワードハッシュを取得"""
        latest = db.query(PasswordHistory)\
            .filter(PasswordHistory.user_id == user_id)\
            .order_by(PasswordHistory.created_at.desc())\
            .first()
        
        if not latest:
            raise HTTPException(status_code=status.HTTP_403_FORBIDDEN, detail="User has no password")
        
        return latest.password

    @staticmethod
    def verify_password_strength(
        new_password: str,
        tenant_id: str,
        db: Session
    ) -> bool:
        """テナントのパスワードポリシーに適合しているか検証"""
        tenant = db.query(Tenant).filter(Tenant.id == tenant_id).first()
        if not tenant:
            raise HTTPException(status_code=status.HTTP_400_BAD_REQUEST, detail="Tenant not found")
        
        # ポリシーチェック（例：最小文字数8文字）
        if len(new_password) < 8:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="Password must be at least 8 characters"
            )
        
        return True

    @staticmethod
    def check_password_history(
        user_id: str,
        new_password: str,
        tenant_id: str,
        db: Session
    ) -> bool:
        """過去パスワードとの重複をチェック"""
        tenant = db.query(Tenant).filter(Tenant.id == tenant_id).first()
        if not tenant:
            raise HTTPException(status_code=status.HTTP_400_BAD_REQUEST, detail="Tenant not found")
        
        pw_histories_limit = tenant.pw_histories_limit
        
        recent_passwords = db.query(PasswordHistory.password)\
            .filter(PasswordHistory.user_id == user_id)\
            .order_by(PasswordHistory.created_at.desc())\
            .limit(pw_histories_limit)\
            .all()
        
        for (hashed,) in recent_passwords:
            if verify_password(new_password, hashed):
                raise HTTPException(
                    status_code=status.HTTP_400_BAD_REQUEST,
                    detail="Password has been used before"
                )
        
        return True

    @staticmethod
    def reset_password(
        user_id: str,
        new_password: str,
        tenant_id: str,
        db: Session
    ) -> None:
        """新しいパスワードを password_histories に追加"""
        hashed_password = hash_password(new_password)
        
        new_history = PasswordHistory(
            tenant_id=tenant_id,
            user_id=user_id,
            password=hashed_password,
        )
        db.add(new_history)
        db.commit()
```

### 7. `backend/app/services/auth_service.py`（新規）

```python
from sqlalchemy.orm import Session
from fastapi import HTTPException, status
from backend.app.models.user import User
from backend.app.models.password_history import PasswordHistory
from backend.app.core.security import create_access_token, verify_password
from backend.app.services.password_service import PasswordService
from backend.app.schemas.auth import AuthResponse

class AuthService:
    @staticmethod
    def login(
        username: str,
        password: str,
        tenant_id: str,
        db: Session
    ) -> AuthResponse:
        """ログイン処理"""
        # ユーザー存在確認
        user = db.query(User)\
            .filter(User.login_id == username, User.tenant_id == tenant_id)\
            .first()
        
        if not user:
            raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Invalid credentials")
        
        # 最新のパスワードを取得して検証
        try:
            current_password_hash = PasswordService.get_current_password_hash(user.id, db)
        except HTTPException:
            raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Invalid credentials")
        
        if not verify_password(password, current_password_hash):
            raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Invalid credentials")
        
        # パスワードリセット要求チェック
        if user.is_required_password_reset:
            return AuthResponse(
                id=user.login_id,
                name=user.name,
                role=user.role,
                groups=[],
                loginStatus="REQUIRES_PASSWORD_RESET",
                reason="INITIAL"
            )
        
        # JWT トークンを生成
        token = create_access_token(user.login_id, tenant_id)
        
        return AuthResponse(
            id=user.login_id,
            name=user.name,
            role=user.role,
            token=token,
            groups=[],
            loginStatus="SUCCESS"
        )

    @staticmethod
    def reset_password(
        login_id: str,
        old_password: str,
        new_password: str,
        tenant_id: str,
        db: Session
    ) -> AuthResponse:
        """パスワードリセット処理"""
        # ユーザー存在確認
        user = db.query(User)\
            .filter(User.login_id == login_id, User.tenant_id == tenant_id)\
            .first()
        
        if not user:
            raise HTTPException(status_code=status.HTTP_403_FORBIDDEN, detail="User not found")
        
        # 旧パスワード検証
        try:
            current_password_hash = PasswordService.get_current_password_hash(user.id, db)
        except HTTPException:
            raise HTTPException(status_code=status.HTTP_403_FORBIDDEN, detail="Invalid old password")
        
        if not verify_password(old_password, current_password_hash):
            raise HTTPException(status_code=status.HTTP_403_FORBIDDEN, detail="Invalid old password")
        
        # 新パスワード強度チェック
        PasswordService.verify_password_strength(new_password, tenant_id, db)
        
        # 過去パスワード重複チェック
        PasswordService.check_password_history(user.id, new_password, tenant_id, db)
        
        # パスワード更新
        PasswordService.reset_password(user.id, new_password, tenant_id, db)
        
        # ユーザーの is_required_password_reset フラグをクリア
        user.is_required_password_reset = False
        db.commit()
        
        # 新 JWT トークンを生成
        token = create_access_token(user.login_id, tenant_id)
        
        return AuthResponse(
            id=user.login_id,
            name=user.name,
            role=user.role,
            token=token,
            groups=[],
            loginStatus="SUCCESS"
        )
```

### 8. `backend/app/routers/auth.py`（新規）

```python
from fastapi import APIRouter, Depends, HTTPException, Header, status, Request
from sqlalchemy.orm import Session
from backend.app.database import get_db
from backend.app.schemas.auth import LoginRequest, PasswordResetRequest, AuthResponse
from backend.app.services.auth_service import AuthService
from backend.app.core.security import get_current_user, create_access_token
from backend.app.models.user import User

router = APIRouter(prefix="/auth", tags=["auth"])

@router.post("/login", response_model=AuthResponse)
async def login(
    request: LoginRequest,
    x_tenant_id: str = Header(..., alias="X-Tenant-ID"),
    db: Session = Depends(get_db)
) -> AuthResponse:
    """ログイン"""
    return AuthService.login(request.username, request.password, x_tenant_id, db)

@router.post("/logout")
async def logout():
    """ログアウト"""
    return {"message": "Logout successful."}

@router.post("/password/reset", response_model=AuthResponse)
async def reset_password(
    request: PasswordResetRequest,
    x_tenant_id: str = Header(..., alias="X-Tenant-ID"),
    db: Session = Depends(get_db)
) -> AuthResponse:
    """パスワードリセット"""
    return AuthService.reset_password(
        request.loginId,
        request.oldPassword,
        request.newPassword,
        x_tenant_id,
        db
    )

# 認証済みルート（/api/auth）
api_router = APIRouter(prefix="/api", tags=["api"])

@api_router.get("/auth", response_model=AuthResponse)
async def get_auth(
    current_user: User = Depends(get_current_user),
    x_tenant_id: str = Header(..., alias="X-Tenant-ID"),
    db: Session = Depends(get_db)
) -> AuthResponse:
    """認証トークン取得（新しいトークンを再発行）"""
    token = create_access_token(current_user.login_id, x_tenant_id)
    
    return AuthResponse(
        id=current_user.login_id,
        name=current_user.name,
        role=current_user.role,
        token=token,
        groups=[],
        loginStatus="SUCCESS"
    )
```

```sql
-- password_histories テーブルに初期パスワードデータを追加
-- admin ユーザーのパスワード：admin@1234（bcrypt ハッシュ化）
-- bcrypt ハッシュ化は FastAPI アプリケーション起動時に自動挿入されるため、

INSERT INTO password_histories (id, tenant_id, user_id, password, created_at)
SELECT 
    gen_random_uuid(),
    t.id,
    u.id,
    '$2b$12$...',  -- bcrypt ハッシュ化されたパスワード
    now()
FROM tenants t
JOIN users u ON u.tenant_id = t.id AND u.login_id = 'admin'
WHERE NOT EXISTS (
    SELECT 1 FROM password_histories WHERE user_id = u.id
);
```

## 実装フロー図

```
POST /auth/login
  ├─ LoginRequest 検証
  ├─ ユーザー存在確認（tenants.id + users.login_id）
  ├─ password_histories から最新パスワードを取得
  ├─ bcrypt で検証
  ├─ is_required_password_reset チェック
  │   ├─ true → REQUIRES_PASSWORD_RESET レスポンス
  │   └─ false → JWT 生成 → SUCCESS レスポンス
  └─ X-Tenant-ID ヘッダー検証

GET /api/auth
  ├─ Authorization ヘッダーから JWT 抽出
  ├─ JWT 検証（署名・有効期限）
  ├─ X-Tenant-ID ヘッダー検証
  ├─ ユーザー情報を取得
  ├─ 新 JWT を生成
  └─ ユーザー情報 + 新 JWT を返す

POST /auth/password/reset
  ├─ PasswordResetRequest 検証
  ├─ ユーザー存在確認
  ├─ 旧パスワード検証
  ├─ 新パスワード強度チェック
  ├─ 過去パスワード重複チェック
  ├─ password_histories に新規レコード追加
  ├─ is_required_password_reset = false に更新
  ├─ 新 JWT 生成
  └─ SUCCESS レスポンス
```

## 注意点

1. **Secret Key 管理**: 本番環境では `SECRET_KEY` を環境変数から読み込む
2. **bcrypt コスト**: パスワードハッシュのコスト設定（デフォルト: 12）をテストで最小化する場合を検討
3. **X-Tenant-ID ヘッダー**: すべてのリクエストで必須。クライアント側で自動付与する仕組みが必要
4. **CORS 設定**: フロントエンドからのリクエストを許可する CORS 設定が必要
