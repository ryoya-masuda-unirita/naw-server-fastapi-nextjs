```python
import pytest
from fastapi import HTTPException
from sqlalchemy.orm import Session

from backend.app.services.password_service import PasswordService
from backend.app.models.password_history import PasswordHistory
from backend.app.core.security import hash_password, verify_password

class TestPasswordService:
    """パスワードサービステスト"""

    async def test_get_current_password_hash_when_exists(self, db_session, test_user_with_password):
        """最新のパスワードハッシュを取得できること"""
        hash_result = PasswordService.get_current_password_hash(
            test_user_with_password["user"].id,
            db_session
        )
        assert hash_result == test_user_with_password["hashed_password"]

    async def test_get_current_password_hash_when_not_exists(self, db_session, test_user):
        """パスワード履歴がない場合、HTTPException 403 を返すこと"""
        with pytest.raises(HTTPException) as exc_info:
            PasswordService.get_current_password_hash(test_user.id, db_session)
        
        assert exc_info.value.status_code == 403

    async def test_verify_password_strength_when_valid(self, db_session, test_tenant):
        """8文字以上のパスワードは検証が通ること"""
        result = PasswordService.verify_password_strength(
            "ValidPassword123",
            test_tenant.id,
            db_session
        )
        assert result is True

    async def test_verify_password_strength_when_too_short(self, db_session, test_tenant):
        """8文字未満のパスワードはHTTPException 400 を返すこと"""
        with pytest.raises(HTTPException) as exc_info:
            PasswordService.verify_password_strength("short", test_tenant.id, db_session)
        
        assert exc_info.value.status_code == 400

    async def test_verify_password_strength_when_tenant_not_found(self, db_session):
        """テナントが存在しない場合、HTTPException 400 を返すこと"""
        with pytest.raises(HTTPException) as exc_info:
            PasswordService.verify_password_strength(
                "ValidPassword123",
                "nonexistent-tenant",
                db_session
            )
        
        assert exc_info.value.status_code == 400

    async def test_check_password_history_when_not_duplicate(self, db_session, test_user_with_password, test_tenant):
        """過去パスワードを使用していない場合、検証が通ること"""
        result = PasswordService.check_password_history(
            test_user_with_password["user"].id,
            "NewPassword123",
            test_tenant.id,
            db_session
        )
        assert result is True

    async def test_check_password_history_when_duplicate(self, db_session, test_user_with_password, test_tenant):
        """過去パスワードを再利用した場合、HTTPException 400 を返すこと"""
        with pytest.raises(HTTPException) as exc_info:
            PasswordService.check_password_history(
                test_user_with_password["user"].id,
                test_user_with_password["plain_password"],
                test_tenant.id,
                db_session
            )
        
        assert exc_info.value.status_code == 400

    async def test_reset_password_creates_new_history(self, db_session, test_user_with_password, test_tenant):
        """新しいパスワードが password_histories に追加されること"""
        new_password = "NewPassword123"
        
        PasswordService.reset_password(
            test_user_with_password["user"].id,
            new_password,
            test_tenant.id,
            db_session
        )
        
        # 最新のパスワードが新しいものであることを確認
        latest_hash = PasswordService.get_current_password_hash(
            test_user_with_password["user"].id,
            db_session
        )
        
        assert verify_password(new_password, latest_hash)
```

## 5. `backend/tests/core/test_security.py`

```python
import pytest
from datetime import datetime, timedelta
import jwt

from backend.app.core.security import (
    create_access_token, decode_token, hash_password, verify_password,
    SECRET_KEY, ALGORITHM
)

class TestSecurityUtilities:
    """JWT・パスワードハッシュユーティリティテスト"""

    async def test_create_and_decode_token(self):
        """JWT エンコード・デコード正常"""
        login_id = "testuser"
        tenant_id = "test-tenant"
        
        token = create_access_token(login_id, tenant_id)
        payload = decode_token(token)
        
        assert payload["sub"] == login_id
        assert payload["tenantId"] == tenant_id

    async def test_decode_expired_token(self):
        """有効期限切れ JWT デコード"""
        payload = {
            "sub": "testuser",
            "tenantId": "test-tenant",
            "exp": datetime.utcnow() - timedelta(hours=1),
            "iat": datetime.utcnow(),
        }
        token = jwt.encode(payload, SECRET_KEY, algorithm=ALGORITHM)
        
        with pytest.raises(HTTPException) as exc_info:
            decode_token(token)
        
        assert exc_info.value.status_code == 401

    async def test_decode_invalid_token(self):
        """不正な JWT デコード"""
        payload = {
            "sub": "testuser",
            "tenantId": "test-tenant",
            "exp": datetime.utcnow() + timedelta(hours=5),
            "iat": datetime.utcnow(),
        }
        token = jwt.encode(payload, "wrong-secret", algorithm=ALGORITHM)
        
        with pytest.raises(HTTPException) as exc_info:
            decode_token(token)
        
        assert exc_info.value.status_code == 401

    async def test_token_contains_correct_payload(self):
        """JWT ペイロード正確性"""
        login_id = "admin"
        tenant_id = "tenant-123"
        
        token = create_access_token(login_id, tenant_id)
        payload = decode_token(token)
        
        assert payload["sub"] == "admin"
        assert payload["tenantId"] == "tenant-123"
        assert "exp" in payload
        assert "iat" in payload

    async def test_hash_and_verify_correct_password(self):
        """bcrypt ハッシュ・検証"""
        password = "TestPassword123!"
        
        hashed = hash_password(password)
        
        assert verify_password(password, hashed)

    async def test_verify_incorrect_password(self):
        """bcrypt 検証失敗"""
        password = "TestPassword123!"
        hashed = hash_password(password)
        
        assert not verify_password("WrongPassword", hashed)

    async def test_hash_produces_different_hash(self):
        """bcrypt 毎回異なるハッシュ生成"""
        password = "TestPassword123!"
        
        hash1 = hash_password(password)
        hash2 = hash_password(password)
        
        assert hash1 != hash2
        assert verify_password(password, hash1)
        assert verify_password(password, hash2)
```
