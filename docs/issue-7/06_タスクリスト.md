### 動作確認（curl）

```bash
# ログイン
curl -X POST http://localhost:8000/auth/login \
  -H "X-Tenant-ID: test-tenant" \
  -H "Content-Type: application/json" \
  -d '{"username":"admin","password":"admin@1234"}'

# レスポンス
{"id":"admin","name":"管理者","role":"ADMIN","token":"eyJ...","groups":[],"loginStatus":"SUCCESS"}

# トークン取得（再発行）
curl -X GET http://localhost:8000/api/auth \
  -H "Authorization: Bearer eyJ..." \
  -H "X-Tenant-ID: test-tenant"

# パスワードリセット
curl -X POST http://localhost:8000/auth/password/reset \
  -H "X-Tenant-ID: test-tenant" \
  -H "Content-Type: application/json" \
  -d '{"loginId":"admin","oldPassword":"admin@1234","newPassword":"newpassword@1234"}'

# 認証なしでアクセス（401）
curl -X GET http://localhost:8000/api/auth \
  -H "X-Tenant-ID: test-tenant"
# 401 Unauthorized
```
